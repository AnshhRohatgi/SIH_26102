-- Table 1: Members of Parliament
CREATE TABLE mp_master (
    mp_id VARCHAR(20) PRIMARY KEY,
    mp_name VARCHAR(100) NOT NULL,
    house_type VARCHAR(20) CHECK (house_type IN ('Lok Sabha', 'Rajya Sabha', 'Nominated')),
    state_constituency VARCHAR(100) NOT NULL,
    tenure_start DATE,
    tenure_end DATE,
    is_active BOOLEAN DEFAULT TRUE
);

-- Table 2: Fund Releases & Allocations
CREATE TABLE fund_allocations (
    allocation_id SERIAL PRIMARY KEY,
    mp_id VARCHAR(20) REFERENCES mp_master(mp_id),
    financial_year VARCHAR(7) NOT NULL,
    entitlement_amount NUMERIC(12,2) DEFAULT 50000000.00,
    released_amount NUMERIC(12,2) DEFAULT 0.00,
    interest_accrued NUMERIC(12,2) DEFAULT 0.00,
    unspent_carried_forward NUMERIC(12,2) DEFAULT 0.00,
    authorization_date DATE,
    cna_reference_no VARCHAR(50)
);

-- Table 3: Work Recommendations & Sanctions
CREATE TABLE work_projects (
    work_id VARCHAR(25) PRIMARY KEY,
    mp_id VARCHAR(20) REFERENCES mp_master(mp_id),
    sector VARCHAR(50) NOT NULL,
    work_title TEXT NOT NULL,
    proposed_cost NUMERIC(12,2) NOT NULL,
    sanctioned_cost NUMERIC(12,2) DEFAULT 0.00,
    quota_type VARCHAR(20) CHECK (quota_type IN ('General', 'SC', 'ST')),
    recommendation_date DATE NOT NULL,
    sanction_date DATE,
    status VARCHAR(30) DEFAULT 'Recommended',
    nodal_district VARCHAR(100),
    executing_agency VARCHAR(100)
);

-- Table 4: Vendor Disbursements
CREATE TABLE vendor_payments (
    bill_id VARCHAR(25) PRIMARY KEY,
    work_id VARCHAR(25) REFERENCES work_projects(work_id),
    vendor_name VARCHAR(100) NOT NULL,
    vendor_gstin VARCHAR(15),
    bill_stage VARCHAR(30),
    claimed_amount NUMERIC(12,2) NOT NULL,
    passed_amount NUMERIC(12,2) NOT NULL,
    net_paid NUMERIC(12,2) NOT NULL,
    pfms_ref_no VARCHAR(40),
    geotag_lat NUMERIC(10,6),
    geotag_long NUMERIC(10,6),
    payment_date DATE,
    payment_status VARCHAR(20) DEFAULT 'Disbursed'
);

-- Table 5: Utilization Certificates
CREATE TABLE utilization_certificates (
    uc_id VARCHAR(25) PRIMARY KEY,
    mp_id VARCHAR(20) REFERENCES mp_master(mp_id),
    nodal_district VARCHAR(100) NOT NULL,
    financial_year VARCHAR(7) NOT NULL,
    authorized_amount NUMERIC(12,2) NOT NULL,
    utilized_amount NUMERIC(12,2) NOT NULL,
    utilization_pct NUMERIC(5,2),
    uc_issue_date DATE,
    verification_status VARCHAR(30) DEFAULT 'Submitted'
);

-- Query 1: Calculate Utilization Rate per MP
SELECT 
    m.mp_name,
    m.state_constituency,
    f.financial_year,
    (f.released_amount + f.unspent_carried_forward) AS total_fund,
    SUM(p.sanctioned_cost) AS total_sanctioned,
    ROUND((SUM(p.sanctioned_cost) / (f.released_amount + f.unspent_carried_forward)) * 100, 2) AS sanction_utilization_pct
FROM mp_master m
JOIN fund_allocations f ON m.mp_id = f.mp_id
LEFT JOIN work_projects p ON m.mp_id = p.mp_id
GROUP BY m.mp_name, m.state_constituency, f.financial_year, f.released_amount, f.unspent_carried_forward;

-- Query 2: Sector-wise MPLADS Spending Analysis
SELECT 
    sector,
    COUNT(work_id) AS total_projects,
    SUM(sanctioned_cost) AS total_approved_amount
FROM work_projects
WHERE status = 'Sanctioned'
GROUP BY sector
ORDER BY total_approved_amount DESC;
